cmake_minimum_required(VERSION 3.20)
project(plc_safety_reference LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(plc_reference src/motor.cpp)
target_include_directories(plc_reference PUBLIC include)
target_compile_options(plc_reference PRIVATE -Wall -Wextra -Wpedantic -Werror)
enable_testing()
add_executable(plc_tests tests/motor_test.cpp)
target_link_libraries(plc_tests PRIVATE plc_reference)
add_test(NAME plc_tests COMMAND plc_tests)
